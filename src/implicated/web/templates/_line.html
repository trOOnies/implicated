{# Line chart drawn as inline SVG. `points` is a list of Point(label, value). #} {% macro line_chart(points, title, width=640, height=180) %} {% set pad_l, pad_r, pad_t, pad_b = 36, 12, 12, 26 %} {% set values = points | map(attribute='value') | list %} {% set top = [(values | max) if values else 0, 1] | max %} {% set top = (top * 1.1) | round(0, 'ceil') %} {% set n = points | length %} {% set step = (width - pad_l - pad_r) / ([n - 1, 1] | max) %} {% set plot_h = height - pad_t - pad_b %} {% set label_every = [(n / 6) | round(0, 'ceil') | int, 1] | max %} {% endmacro %}
